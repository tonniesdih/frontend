
CREATE DATABASE IF NOT EXISTS kiambuhighform;

USE kiambuhighform;

CREATE TABLE IF NOT EXISTS students (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    age INT,
    student_class VARCHAR(30),
    admission_number VARCHAR(30) UNIQUE,
    stream VARCHAR(30)
);
